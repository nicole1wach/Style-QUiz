function doPost(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const data = JSON.parse(e.postData.contents); if (sheet.getLastRow() === 0) { sheet.appendRow([ "Timestamp", "Assigned Identity", "Agreement Score", "Own Style Description", "What Algorithm Missed", "Would Real Algorithm Match?" ]); } sheet.appendRow([ new Date().toISOString(), data.identity, data.agreeScore, data.ownStyle, data.missed, data.realAlgo ]); return ContentService .createTextOutput(JSON.stringify({ status: "success" })) .setMimeType(ContentService.MimeType.JSON); }
